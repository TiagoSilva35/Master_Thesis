% ===================================================== 4. BAYESIAN SPLINES 1/2
\pageset{Bayesian splines}{4}
\begin{frame}
  \ctitle{Bayesian splines (1/2): believability before and after the data}
  \nblock{0.85}{1.98}{5.55}{1}{Believability before the data}{%
    $\text{believability}(h)\ \propto\ \exp\!\big(-\tfrac{\lambda}{2\sigma^2}\sum_k (h_{k-1}-2h_k+h_{k+1})^2\big)$\\[3pt]
    Heights 1, 2, 3 do not bend (0): most believable. Heights 1, 3, 1 bend by $-4$: cost $16\cdot\tfrac{\lambda}{2\sigma^2}$. $\lambda$ is how strongly we believe the curve is smooth.}
  \nblock{0.85}{3.75}{5.55}{2}{After the observations}{%
    believability after $\propto$ fit to the data $\times$ believability before.\\[3pt]
    Most believable heights: $\;(W^\top W+\lambda S)\,h=W^\top y$\\[2pt]
    {\color{grey}$W$: each person's bump weights (one row each) \textperiodcentered{} $S$: the bending cost}}
  \nblock{0.85}{5.45}{5.55}{3}{How unsure the curve is}{%
    $\Sigma=\sigma^2\,(W^\top W+\lambda S)^{-1}$ gives each height's spread.\\[2pt]
    Band $=$ curve $\pm\,1.96$ spreads (about 95\% of believable curves).}
  \fig{6.95}{2.15}{5.6}{prior_posterior.pdf}
  \tb{5.4}{7.05}{4.85}{\color{grey}\fs{12.5}{15.5}Left: random curves that are believable before any data (no bending is cheap). Right: random curves still believable after 18 people; together they form the band.\par}
\end{frame}

% ===================================================== 5. BAYESIAN SPLINES 2/2
\pageset{Bayesian splines}{5}
\begin{frame}
  \ctitle{Bayesian splines (2/2): choosing $\lambda$ and the noise size}
  \nblock{0.85}{1.98}{5.55}{1}{Evidence: how probable is the data under $\lambda$?}{%
    Averaged over every believable curve.
    Too small $\lambda$: many wiggly curves share the believability, so each fits little. Too large: only straight lines, which miss the data.\\[3pt]
    $\log\text{evidence}=-\tfrac{n}{2}\log\sigma^2+\tfrac12\log|\lambda S|-\tfrac12\log|W^\top W+\lambda S|$}
  \nblock{0.85}{4.0}{5.55}{2}{Noise size $\sigma^2$}{%
    The average square of what the curve leaves unexplained:\\[2pt]
    $\sigma^2=\big(\sum_i(y_i-\hat y_i)^2+\lambda\,h^\top S h\big)/n$\\[2pt]
    Leftovers $0.3,-0.2,0.1$: $\sigma^2\approx(0.09+0.04+0.01)/3=0.047$.}
  \nblock{0.85}{5.6}{5.55}{3}{One $\lambda$ per curve}{%
    Grid from 0.1 to $10^4$; each curve's $\lambda$ is tuned in turn, three rounds. For Y (a chance, not a bell): Newton's method and a Laplace approximation.}
  \fig{6.9}{2.05}{5.7}{evidence.pdf}
\end{frame}
