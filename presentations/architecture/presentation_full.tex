  \tb{3.6}{8.85}{3.15}{\color{body}\fs{14.5}{18.5}Score the simulated person with the classifier's curves.}
  \hrl{0.85}{4.25}{11.6}
  \tb{11.6}{0.85}{4.45}{\color{grey}\fs{15}{19}Repeat for each candidate action, and keep the cheapest that passes:}
  \tb{11.6}{0.85}{4.85}{\color{ink}\fs{21}{26}$\min_a\ \mathsf{cost}(a;\mathbf{x})\quad\text{subject to}\quad P(Y{=}1\mid\mathbf{x}^{\mathsf{after}}(a))\geq 1-\varepsilon$}
  \tb{11.6}{0.85}{5.48}{\color{body}\fs{15}{19}{\bfseries\color{ink}Kind of system:} an optimisation solver, run once per rejected person. It has no training and no curves of its own, so it cannot drift by itself.\par}
  \slidefoot{Built on abduction, action, prediction (Karimi et al.\ 2021), lower-confidence recourse (Karimi et al.\ 2020)\\ and improvement (K\"onig et al.\ 2023). Not novel or adapted.}
\end{frame}
