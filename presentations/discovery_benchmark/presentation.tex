% ===================================================== 6. THE EVIDENCE
\pageset{The score}{6}
\begin{frame}
  \ctitle{The score behind the search: the evidence}
  \tb{11.6}{0.85}{1.95}{\color{ink}\fs{22}{26}$\log\text{evidence}(x_j \mid \text{parents}) \;=\;
    \underbrace{{\color{cred}-\tfrac{N}{2}\log\hat\sigma^2}}_{\textstyle\text{fit}}
    \;+\; \underbrace{\tfrac12\log|P| \;-\; \tfrac12\log|W^\top W+P|}_{\textstyle\text{flexibility cost}}$\par}
  \lowcol{0.85}{3.3}{3.6}{\color{cred}Fit}{$\hat\sigma^2$: the average squared leftover (actual $-$ curves). Smaller leftovers, higher score. With $N=1{,}000$: $\hat\sigma^2=1$ gives 0 points, $\hat\sigma^2=0.2$ about 800.}
  \lowcol{4.85}{3.3}{3.6}{Flexibility cost}{$W$: each row's bump weights. $P=\lambda\,\times$ bending, the smoothness belief. More parents or wigglier curves cost more.}
  \lowcol{8.85}{3.3}{3.6}{In the search}{A graph's score is the sum over its variables. The gain of $i\rightarrow j$ is $j$'s evidence with $i$ minus without.}
  \colhead{0.85}{5.0}{11.6}{Two assumptions are built in}
  \tb{11.6}{0.85}{5.62}{\color{body}\fs{15}{20}%
    \textbf{\color{ink}1 \ Bell-shaped noise:} the fit term is the bell curve's formula; only the size $\hat\sigma^2$ enters, never the shape.\\
    \textbf{\color{ink}2 \ Numbers on a line:} $W$ and $P$ build a spline over every parent, so neighbouring values must have similar effects.\par}
\end{frame}

% ===================================================== 7. ISSUES
\pageset{Issues}{7}
\begin{frame}
  \ctitle{Two issues found}
  \colhead{0.85}{1.98}{5.6}{1 \textperiodcentered{} The noise model (the fit term)}
  \lowcol{0.85}{2.62}{5.6}{Roots and noise that are not bell-shaped}{The fit term sees only the size $\hat\sigma^2$, not the shape, so a reversed arrow can win.}
  \lowcol{0.85}{3.72}{5.6}{Categorical children}{A 0/1 child leaves $-0.3$ or $+0.7$, never a bell: $\hat\sigma^2$ no longer measures the fit.}
  \lowcol{0.85}{4.82}{5.6}{Near-zero noise}{spread $=$ note rate $-$ index, rounded: $\hat\sigma^2=0.0000035$, so $-\tfrac{N}{2}\log\hat\sigma^2=6{,}286$ points. Its rearrangements score 6,050 and 5,811.}
  \colhead{7.0}{1.98}{5.45}{2 \textperiodcentered{} The input model ($W$ and $P$)}
  \lowcol{7.0}{2.62}{5.45}{Nominal parents}{Event type coded 0--3: $W$ and $P$ invent an order, so renumbering changes the evidence.}
  \lowcol{7.0}{3.72}{5.45}{Ordinal parents}{Credit band has 5 values but 11 heights: most of them are set by $P$ alone.}
  \lowcol{7.0}{4.82}{5.45}{Binary parents}{Harmless: two heights, one effect.}
\end{frame}
