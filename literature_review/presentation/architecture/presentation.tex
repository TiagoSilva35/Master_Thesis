% ===================================================== STEP 1
\pageset{Updates}{15}
\begin{frame}
  \steptitle{1}{every sample}{Collect the evidence}
  \tb{11.6}{0.85}{2.05}{\centering\color{ink}\fs{23}{28}$A_j\leftarrow\gamma A_j+\Phi_j\Phi_j^{\top}\qquad\qquad b_j\leftarrow\gamma b_j+\Phi_j x_j$\par}
  \lowcol{0.85}{2.95}{3.6}{$A$: what was seen}{One entry per pair of coefficients: how often recent samples activated both.}
  \lowcol{4.85}{2.95}{3.6}{$b$: what was observed}{Each sample's bump values, weighted by the value observed.}
  \lowcol{8.85}{2.95}{3.6}{$\gamma$: how long it lasts}{Memory is about $1/(1-\gamma)$ samples: $\gamma=0.998$ keeps about 500.}
  \hrl{0.85}{4.3}{11.6}
  \lowcol{0.85}{4.5}{5.5}{Why this is enough}{The least-squares fit is $\hat\theta_j=A_j^{-1}b_j$, so raw data is not stored.}
  \lowcol{6.95}{4.5}{5.5}{Fixed cost}{Size depends on the number of coefficients, not on the length of the stream.}
  \tb{11.6}{0.85}{5.7}{\color{ink}\bfseries\fs{16}{20}The curves do not move in this step.\par}
  \slidefoot{Exact for feature nodes, which are linear in their coefficients.}
\end{frame}

% ===================================================== STEP 2
\pageset{Updates}{16}
\begin{frame}
  \steptitle{2}{every sample}{Read the score}
  \tb{11.6}{0.85}{1.98}{\color{ink}\fs{18}{22}$s_j=b_j-A_j\theta_j=\sum_r\gamma^{\,t-r}\,\Phi_r\,(x_{j,r}-\Phi_r^{\top}\theta_j)$\quad{\color{body}\fs{14.5}{18}$r$ is a past sample. One block $s_e$ per edge.}\par}
  \scorepanel{0.85}{{.25,-.3,.2,.3,-.25,.3,-.2,.25,-.3,.15}}{Curve still right}{Errors have no pattern and cancel.}{grey}
  \scorepanel{6.95}{{.2,-.15,.3,2.2,2.6,2.3,.3,-.2,.2,-.15}}{Curve wrong in one range}{Errors share a sign there and add up.}{cred}
  \tb{11.6}{0.85}{5.75}{\color{ink}\bfseries\fs{16}{20}Small $\|s_e\|$: the sample ends here. Most samples do.\par}
  \slidefoot{Schematic: the score of each coefficient on one edge.}
\end{frame}

% ===================================================== STEP 3
\pageset{Updates}{17}
\begin{frame}
  \steptitle{3}{when a score passes the threshold}{Decide: one rule detects and updates}
  \tb{11.6}{0.85}{2.05}{\centering\color{ink}\fs{23}{28}$\min_{\delta}\ \ \tfrac12\delta^{\top}A\delta-\delta^{\top}s\ +\ \lambda\sum_e\|\delta_e\|\ +\ \lambda_2\|\delta\|_1$\par}
  \lowcol{0.85}{2.95}{3.6}{Fit: $\tfrac12\delta^{\top}A\delta-\delta^{\top}s$}{Fall in squared error, from $A$ and $s$ alone.}
  \lowcol{4.85}{2.95}{3.6}{Edge penalty: $\lambda\sum_e\|\delta_e\|$}{Paid per edge, so most edges stay at zero.}
  \lowcol{8.85}{2.95}{3.6}{Local penalty: $\lambda_2\|\delta\|_1$}{Paid per coefficient, so the edit stays local.}
  \hrl{0.85}{4.1}{11.6}
  \tb{11.6}{0.85}{4.3}{\color{ink}\bfseries\fs{21}{26}$\delta=0$ is optimal exactly while every edge has $\|s_e\|\leq\lambda$.\par}
  \tb{11.6}{0.85}{4.85}{\color{body}\fs{15}{19}The change test is the fit's optimality condition, so no separate drift detector is needed. Below $\lambda$, neither noise nor a small real change moves a curve.\par}
  \slidefoot{Strictly, $\|s_e\|$ with the mean error removed and each coefficient score shrunk by $\lambda_2$.\\ Group lasso (Yuan \& Lin 2006), sparse-group lasso (Simon et al.\ 2013), online sparse estimation (Angelosante et al.\ 2010).\\ Per-edge use in an online causal model: possibly novel.}
\end{frame}

% ===================================================== STEP 4
\pageset{Updates}{18}
\begin{frame}
  \steptitle{4}{only after a decision}{Commit}
  \chain{0.85}{1}{Keep the chosen}{The set $S$ of coefficients the fit left non-zero.}
  \chain{3.85}{2}{Refit them}{$\delta_S=A_{SS}^{-1}s_S$, with no penalty, because penalties shrink the edit.}
  \chain{6.85}{3}{Save a version}{$\phi_e^{t}=\phi_e^{t-1}+\delta_e^{t}$, so any past curve can be rebuilt.}
  \chain{9.85}{4}{Record a Change}{Edge, knot range, $\delta$, version and time.}
  \hrl{0.85}{4.2}{11.6}
  \tb{11.6}{0.85}{4.42}{\color{body}\fs{15}{19}On $S$ the score falls to zero, because the new curve fits the statistics. $A$ and $b$ are unchanged.\par}
  \tb{11.6}{0.85}{4.95}{\color{body}\fs{15}{19}At the first commit the statistics still hold old data, so one real change can take several commits on the same edge.\par}
  \slidefoot{Refitting after selection: Belloni \& Chernozhukov 2013. Sparse versions of spline coefficients: adapted.}
\end{frame}
